% Command to check if string is contained in list of strings
\usepackage{xstring}
\newcommand\IfStringInList[2]{\IfSubStr{,#2,}{,#1,}}

% Degree course and faculty
\newcommand{\DegreeCourse}[1]{
	% Set name of degree program
	\ifthenelse{\equal{#1}{EuI}}{\def\DegreeName{Bachelor of Science Elektrotechnik und Informationstechnik}}{}
	\ifthenelse{\equal{#1}{REE}}{\def\DegreeName{Bachelor of Science Regenerative Energiesysteme und Energiemanagement – Elektro- und Informationstechnik}}{}
	\ifthenelse{\equal{#1}{IE}}{\def\DegreeName{Bachelor of Science Information Engineering}}{}
	\ifthenelse{\equal{#1}{MT}}{\def\DegreeName{Bachelor of Science Medientechnik}}{}
	\ifthenelse{\equal{#1}{M-A}}{\def\DegreeName{Master of Science Automatisierung}}{}
	\ifthenelse{\equal{#1}{M-ICE}}{\def\DegreeName{Master of Science Information and Communications Engineering}}{}
	\ifthenelse{\equal{#1}{M-MES}}{\def\DegreeName{Master of Science Microelectronic Systems}}{}
	
	% Set constant names for Bachelor's and Master's degrees
	\IfStringInList{#1}{EuI,REE,IE,MT}{
		\DE{\newcommand{\ThesisKind}{Bachelorarbeit}}{\newcommand{\ThesisKind}{Bachelor's thesis}}
		\DE{\newcommand{\ExamKind}{Bachelorprüfung}}{\newcommand{\ExamKind}{examination and study regulations}}
	}{
		\DE{\newcommand{\ThesisKind}{Masterarbeit}}{\newcommand{\ThesisKind}{Master's thesis}}
		\DE{\newcommand{\ExamKind}{Masterprüfung}}{\newcommand{\ExamKind}{examination and study regulations}}
	}
	
	% Set faculty
	\IfStringInList{#1}{EuI,REE,IE,MT,M-A,M-ICE,M-MES}{
		\def\FacultyDE{Fakultät Elektro-, Medien- und Informationstechnik}
		\def\FacultyEN{Faculty of Electrical Engineering, Media and Information Technology}
	}{
	}
	
	% Set source files for titles
	\ifthenelse{\equal{#1}{M-MES}}{
		\def\TitleFront{\input{SpecialPages/Title_Front_MES}\cleardoublepage}
		\def\TitleInside{\input{SpecialPages/Title_Inside_MES}\cleardoublepage}
	}{
		\def\TitleFront{\input{SpecialPages/Title_Front}\cleardoublepage}
		\def\TitleInside{\input{SpecialPages/Title_Inside}\cleardoublepage}
	}
}
